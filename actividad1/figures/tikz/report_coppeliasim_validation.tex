\begin{tikzpicture}[
  % --- ESTILOS GLOBALES ---
  card/.style={
    draw=UniPrimaryDeep!40, fill=white, rounded corners=3mm,
    minimum width=5.8cm, minimum height=2.1cm, inner sep=0pt, % Dimensiones ampliadas
    drop shadow={opacity=0.03, shadow xshift=1pt, shadow yshift=-1pt}
  },
  num/.style={
    circle, fill=UniPrimary, text=white, font=\small\bfseries\sffamily,
    minimum size=0.6cm, inner sep=0pt, align=center
  },
  arr/.style={
    -{Latex[length=2.5mm,width=1.8mm]}, line width=1.2pt, draw=UniPrimaryDeep!60,

  % --- MACRO MEJORADA (Eje central dividido) ---
  \def\stepcard#1#2#3#4#5#6{
    \node[card, #2] (#1) {};
    
    % Círculo con el número (esquina superior izquierda)
    \node[num, fill=#3, anchor=north west] 
      (#1-n) at ($(#1.north west) + (12pt, -14pt)$) {#4};
    
    % Título (Anclado hacia ARRIBA desde el centro del círculo)
    \node[anchor=south west, font=\normalsize\bfseries\sffamily, text=UniPrimaryDeep, inner sep=0pt] 
      (#1-t) at ($(#1-n.east) + (8pt, 2pt)$) {#5};
      
    % Descripción (Anclada hacia ABAJO desde el centro del círculo)
    % El 'inner sep=0pt' y 'text width=4.1cm' aseguran que no rebase la caja
    \node[anchor=north west, font=\footnotesize\sffamily, text=UniDark, text width=4.1cm, align=left, inner sep=0pt] 
      (#1-d) at ($(#1-n.east) + (8pt, -2pt)$) {#6};
  }

  % --- CREACIÓN DE LAS TARJETAS ---
  
  % 1. Ruta Base

  % --- FLECHAS DE FLUJO ---
  \draw[arr] (c1.east) -- (c2.west);
  \draw[arr] (c2.south) -- (c4.north);
  \draw[arr] (c4.west) -- (c3.east);
  \draw[arr] (c3.north) -- (c1.south);

  % --- FOOTER ---
  \node[anchor=north, font=\footnotesize\bfseries\sffamily, text=UniPrimaryDeep, align=center,
