Profesor: Jose I. I\~niguez\\

\slide{Objetivo de la propuesta}{
\textbf{Alcance:} mejorar logistica interna, trazabilidad e inspeccion FOD en el entorno HTP A320 y seccion 19.1 de Alestis Puerto Real.
\begin{itemize}
  \item No se propone fabricar el HTP con robots moviles.
  \item Automatizacion de kitting, herramientas, consumibles, retorno de utiles y registros.
  \item Apoyo al escalado desde 40 HTP/mes hacia 80 HTP/mes o mas.
\end{itemize}
}

\slide{Presentacion de Lunabotics}{
\begin{itemize}
  \item Empresa ficticia joven de integracion de robotica movil industrial.
  \item Capacidades: simulacion CoppeliaSim, AMR, sensores LiDAR/RGB-D/RFID, HMI y datos de planta.
  \item Experiencia de referencia: pilotos intralogisticos, celdas con bases omnidireccionales y trazabilidad QR/RFID.
  \item Enfoque comercial: piloto medible, expansion controlada y despliegue extendido solo con KPIs validados.
\end{itemize}
}

\slide{Contexto: Alestis, HTP y seccion 19.1}{
\begin{itemize}
  \item El HTP es el estabilizador horizontal de cola del A320.
  \item La seccion 19.1 se asocia al entorno posterior e interfaces de cola.
  \item El proceso requiere utiles, kits, herramientas, consumibles, inspecciones y registros.
  \item La oportunidad esta en estabilizar el flujo operativo, no en cambiar el montaje certificado.
\end{itemize}
}

\slide{Problemas detectados}{
\begin{itemize}
  \item Desplazamientos improductivos de operarios.
  \item Esperas por kits, herramientas y consumibles.
  \item Riesgo FOD en zonas de trabajo aeronautico.
  \item Trazabilidad fina insuficiente.
  \item Coordinacion manual entre estaciones y logistica.
  \item Necesidad de escalar capacidad sin aumentar desorden operativo.
\end{itemize}
}

\slide{Robot seleccionado}{
\textbf{AMR omnidireccional}: plataforma tipo KUKA YouBot, KUKA Omnirob o equivalente industrial representable en CoppeliaSim.
\begin{itemize}
  \item Ruedas mecanum/suecas: movimiento lateral sin cambiar orientacion.
  \item Mejor maniobra cerca de estaciones, estructuras y operarios.
  \item Mas flexible que un AGV de ruta fija.
  \item Adecuado para validar navegacion, sensores, actuadores y misiones.
\end{itemize}
}

\slide{Sensores y arquitectura}{
\begin{itemize}
  \item LiDAR 2D/3D para SLAM, obstaculos y seguridad.
  \item Camara RGB-D para inspeccion FOD.
  \item RFID/QR para kits, herramientas y entregas.
  \item IMU, encoders, bumpers, escaner laser y E-stop.
  \item Planificacion global A*/Dijkstra y evitacion local DWA/campos potenciales como referencia.
  \item HMI, WiFi industrial, supervisor y base de datos.
\end{itemize}
}

\slide{Flujo operativo}{
\begin{enumerate}
  \item Planificacion de mision.
  \item Carga de kit o herramienta.
  \item Lectura RFID/QR.
  \item Navegacion hasta estacion.
  \item Entrega y confirmacion.
  \item Inspeccion FOD si aplica.
  \item Retorno de utiles o nueva mision.
\end{enumerate}
}

\slide{Escenario piloto: 1 robot}{
\begin{itemize}
  \item Objetivo: validar navegacion, seguridad, kitting e inspeccion FOD.
  \item Alcance: zona acotada y estaciones seleccionadas.
  \item Riesgos: aceptacion de usuarios, mapa incompleto, interferencias.
  \item Criterios: 95\% de misiones completadas, cero incidentes, esperas reducidas.
  \item Coste estimado: 198.700 \texteuro{}.
\end{itemize}
}

\slide{Viabilidad y beneficios}{
\begin{itemize}
  \item Seguridad mediante sensores, zonas, velocidad limitada y E-stop.
  \item Integracion gradual: HMI local, base de datos, despues MES/ERP.
  \item Beneficios: menor espera, trazabilidad, datos de flujo y prevencion FOD.
  \item Validacion previa en CoppeliaSim: rutas, obstaculos, aproximacion lateral y misiones trazadas.
  \item Riesgos mitigados: aceptacion de operarios, mapa desactualizado, WiFi y lecturas RFID/QR.
\end{itemize}
}

\slide{Conclusion}{
\textbf{Decision recomendada:} iniciar con piloto de 1 AMR omnidireccional, validado en CoppeliaSim y medido con KPIs de seguridad, entregas, trazabilidad, FOD y aceptacion operativa.
\begin{itemize}
  \item Solucion viable, escalable y realista.
  \item Alineada con robotica movil industrial.
  \item Apoya el aumento de capacidad sin prometer automatizacion imposible.
\end{itemize}
